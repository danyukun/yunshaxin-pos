## 一、基础参数类型

```typescript
// 颜色项（参数设置 → 颜色管理）
interface ColorItem {
  id: string;
  name: string;   // 如"黑色"
  hex: string;    // 如"#1a1a1a"（花色 hex 为空字符串）
}

// 简单命名项（品牌、瑕疵、处理结果等）
interface SimpleItem {
  id: string;
  name: string;
}

// 带价格项（附件、工艺）
interface PricedItem {
  id: string;
  name: string;
  price: number;  // 单位：元
}

// 衣物大品类
interface Category {
  id: string;   // 单字母：A/B/C/D/E/F/G/H/R
  code: string; // 同 id
  name: string; // 如"上衣外套"
  color: string; // 品类标识色（hex）
}
```

interface CatalogItem {

## 三、会员体系

```typescript
interface Member {
  id: string;
  name: string;
  phone: string;               // 唯一标识
  balance: number;             // 账户余额（元）
  totalSpent: number;          // 累计消费（元，用于等级计算）
  registrationDate: string;    // 注册日期 'YYYY-MM-DD'
  addresses: MemberAddress[];  // 地址簿（支持多地址）
  coupons: MemberCoupon[];     // 持有优惠券
  isImportant?: boolean;       // 重要客户标记
  importantLevel?: 1 | 2 | 3; // 重要程度（1=⭐ 2=⭐⭐ 3=⭐⭐⭐）
  isFollowWechat?: boolean;    // 是否关注微信
  storeId?: string;            // 主要消费门店
}

interface MemberAddress {
  id: string;
  label: string;     // 如"家"、"公司"
  address: string;   // 详细地址文字
  phone: string;     // 联系电话
  isDefault: boolean;
}

interface MemberCoupon {
  id: string;
  name: string;       // 如"满100减20"
  amount: number;     // 抵扣金额（元）
  minSpend: number;   // 最低消费门槛（0=无门槛）
  expiry: string;     // 有效期 'YYYY-MM-DD'
}

interface MemberTier {
  id: string;
  name: string;           // 如"金卡会员"
  minSpent: number;       // 达到此等级的最低累计消费（元）
  discountRate: number;   // 折扣率（1.0=无折扣 0.85=85折）
  color: string;          // 等级标识色
  badge: string;          // Tailwind CSS class
}
```

### 默认会员等级配置

| 等级 | 门槛 | 折扣 |
|------|------|------|
| 普通会员 | ¥0+ | 无折扣 |
| 银卡会员 | ¥500+ | 9.5折 |
| 金卡会员 | ¥1500+ | 9折 |
| 黑卡会员 | ¥5000+ | 8.5折 |

## 四、衣物状态机（20态）

```typescript
type GarmentStatus =
  | 'ordered'          // 客户下单
  | 'assigned'         // 预约单分配（C型上门取）
  | 'picking_up'       // 取衣中（C型）
  | 'pickup_done'      // 取衣完成（C型）
  | 'received'         // 已收衣（到店）
  | 'sent_wash'        // 已送洗（发出批次）
  | 'factory_in'       // 已入厂
  | 'factory_sorted'   // 工厂分拣
  | 'washing'          // 洗涤中
  | 'initial_check'    // 初检
  | 'ironing'          // 熨烫
  | 'final_check'      // 总检
  | 'packing'          // 打包
  | 'factory_out'      // 已出厂
  | 'store_in'         // 已入库（到店待上架）
  | 'on_shelf'         // 已上架（挂点就位，待取）
  | 'delivering'       // 配送中（B/C型）
  | 'completed'        // 已完成
  | 'cancelled'        // 已退单
  | 'compensated';     // 已退赔
```
